### Markdown links

```bash
ruby -EUTF-8:UTF-8 -e '
file = ARGV.fetch(0)
missing = []
File.read(file, encoding: "UTF-8")
  .scan(/\[[^\]]+\]\(([^)]+)\)/)
  .flatten
  .each do |raw|
    next if raw.start_with?("http://", "https://", "mailto:", "#")
    target = raw.sub(/\A</, "").sub(/>\z/, "").split("#", 2).first
    next if target.empty?
    path = File.expand_path(target, File.dirname(file))
    missing << raw unless File.exist?(path)
  end
abort("missing links: #{missing.join(", ")}") unless missing.empty?
' "$spec"
```
